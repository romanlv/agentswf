import { spawn } from "node:child_process";
import { createInterface } from "node:readline";
import { readFileSync, writeFileSync } from "node:fs";
import { Glob } from "bun";

const MODEL = "gpt-6-luna";
const cwd = process.cwd();

function server() {
  const p = spawn("codex", ["app-server", "--listen", "stdio://", "-c", 'sandbox_mode="danger-full-access"'], { cwd, stdio: ["pipe", "pipe", "inherit"] });
  const rl = createInterface({ input: p.stdout });
  let next = 10;
  const waiting = new Map<number, (r: any) => void>();
  const listeners: ((r: any) => void)[] = [];
  rl.on("line", (line) => {
    let r: any; try { r = JSON.parse(line); } catch { return; }
    if (r.id !== undefined && waiting.has(r.id)) { waiting.get(r.id)!(r); waiting.delete(r.id); }
    for (const l of listeners) l(r);
  });
  const call = (method: string, params: object) => new Promise<any>((res) => {
    const id = next++; waiting.set(id, res);
    p.stdin.write(JSON.stringify({ jsonrpc: "2.0", id, method, params }) + "\n");
  });
  const turn = (threadId: string, text: string) => new Promise<any>((res) => {
    const usages: string[] = []; let answer = "";
    const l = (r: any) => {
      if (r.method === "thread/tokenUsage/updated" && r.params.threadId === threadId) { const u = r.params.tokenUsage.last; usages.push(`${u.inputTokens - u.cachedInputTokens}/${u.cachedInputTokens}`); }
      if (r.method === "item/completed" && r.params?.threadId === threadId && r.params?.item?.type === "agentMessage") answer = r.params.item.text;
      if (r.method === "turn/completed" && r.params.threadId === threadId) { listeners.splice(listeners.indexOf(l), 1); res({ answer, requests: usages }); }
    };
    listeners.push(l);
    call("turn/start", { threadId, input: [{ type: "text", text }], model: MODEL });
  });
  return { init: async () => { await call("initialize", { clientInfo: { name: "awf", version: "0" } }); p.stdin.write(JSON.stringify({ jsonrpc: "2.0", method: "initialized" }) + "\n"); }, call, turn, close: () => new Promise<void>((res) => { p.once("exit", () => res()); p.stdin.end(); setTimeout(() => p.kill(), 3000); }) };
}

function rollout(id: string) {
  const home = `${process.env.HOME}/.codex/sessions`;
  for (const f of new Glob(`**/rollout-*${id}.jsonl`).scanSync(home)) return `${home}/${f}`;
  throw new Error("no rollout " + id);
}

const manual = Array.from({ length: 900 }, (_, i) => `Section ${i}: the valve ${i * 7 % 113} opens when gauge ${i % 17} reads ${(i * 37) % 1000} kPa, unless relay R${i % 29} is latched.`).join("\n");
const log: any = {};
const a = server(); await a.init();
const started = await a.call("thread/start", { model: MODEL, cwd });
const parent = started.result.thread.id;
log.parent = parent;
log.parentTurn = await a.turn(parent, `Here is a manual.\n\n${manual}\n\nRemember the codename OSPREY. Reply with just "noted".`);
log.parentTurn2 = await a.turn(parent, `Which relay goes with section 412? Answer briefly.`);
const forks: Record<string, string> = {};
for (const name of ["control", "patched"]) {
  const f = await a.call("thread/fork", { threadId: parent, excludeTurns: true });
  if (f.error) throw new Error(JSON.stringify(f.error));
  forks[name] = f.result.thread.id;
}
await a.close();
const parentSession = JSON.parse(readFileSync(rollout(parent), "utf8").split("\n")[0]).payload.session_id;
log.parentSession = parentSession;
const file = rollout(forks.patched!);
const lines = readFileSync(file, "utf8").split("\n");
const meta = JSON.parse(lines[0]!); log.patchedBefore = meta.payload.session_id; meta.payload.session_id = parentSession;
lines[0] = JSON.stringify(meta); writeFileSync(file, lines.join("\n"));
log.forks = forks;

const order = process.argv[2] === "control-first" ? ["control", "patched"] : ["patched", "control"];
for (const name of order) {
  const b = server(); await b.init();
  const r = await b.call("thread/resume", { threadId: forks[name] });
  if (r.error) { log[name] = { error: r.error }; await b.close(); continue; }
  log[name] = await b.turn(forks[name]!, "What codename did I tell you? Answer with just the word.");
  log[name].second = await b.turn(forks[name]!, "And which relay goes with section 13? Briefly.");
  await b.close();
}
console.log(JSON.stringify(log, null, 2));
process.exit(0);
